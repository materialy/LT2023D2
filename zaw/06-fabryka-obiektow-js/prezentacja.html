function createTv(brand, screenSize, color) {
const colors = ["white", "black", "silver"];
if (!colors.includes(color)) {
return null;
}
return { brand, screenSize, color };
}
const tv = createTv("Lumo", 42, "black");
if (tv === null) console.log("Brak telewizora");
else console.log(tv.brand + ", " + tv.screenSize + " cali");

Do zapamiętania

  • Fabryka to funkcja, która buduje i return-uje nowy obiekt.
  • Każde wywołanie fabryki daje nowy, osobny obiekt.
  • { brand } to skrót od { brand: brand }, tylko przy tej samej nazwie.
  • tablica.includes(x) daje true albo false; porównuje jak ===, z wielkością liter.
  • ! odwraca wynik: !colors.includes(color) znaczy „koloru nie ma na liście”.
  • Walidacja na początku funkcji: złe dane, to od razu return null.
  • Klamra obiektu w tej samej linii co return, inaczej wynik to undefined.
  • Przed odczytem tv.brand sprawdź, czy tv !== null; tablicę wyników oczyścisz przez filter.
